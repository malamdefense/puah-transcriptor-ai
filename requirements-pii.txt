# תלויות הסרת PII (HebSafeHarbor) — מותקנות בסביבה וירטואלית *מבודדת* (.venv-pii)
# בלבד, כי הן דורשות pydantic<2 / spaCy 3.2 שמתנגשים עם האפליקציה הראשית.
# ה-worker קורא להן כתת-תהליך דרך app/pii_runner.py.
#
# התקנה ידנית:
#   python -m venv .venv-pii
#   .venv-pii\Scripts\python -m pip install -r requirements-pii.txt
# (או פשוט: python rebuild.py — מתקין גם את זה אוטומטית)

# spaCy 3.2 / thinc / blis הוותיקים נבנו מול numpy 1.x — numpy 2 שובר אותם
# (numpy.dtype size changed). חובה להצמיד מתחת ל-2.
numpy<2

# presidio-analyzer ננעל ל-2.2.31 (הגרסה האחרונה התומכת ב-spaCy 3.2). חובה לנעול
# גם את ה-anonymizer לאותה גרסה — אחרת ה-operator המותאם 'replace_in_hebrew'
# של HebSafeHarbor אינו מזוהה (InvalidParamError).
presidio-analyzer==2.2.31
presidio-anonymizer==2.2.31

hebsafeharbor
