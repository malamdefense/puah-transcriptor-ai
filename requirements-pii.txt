# PII removal dependencies (HebSafeHarbor) — installed in isolated virtual environment (.venv-pii)
# only, because they require pydantic<2 / spaCy 3.2 that conflicts with the main app.
# The worker calls them as a sub-process via app/pii_runner.py.
#
# Manual installation:
#   python -m venv .venv-pii
#   .venv-pii\Scripts\python -m pip install -r requirements-pii.txt
# (or simply: python rebuild.py — it installs this automatically)

# spaCy 3.2 / thinc / blis old versions built with numpy 1.x — numpy 2 breaks them
# (numpy.dtype size changed). Must be locked below 2.
numpy<2

# presidio-analyzer locked to 2.2.31 (the latest version supporting spaCy 3.2). Must be locked
# also to the same version — otherwise the 'replace_in_hebrew' operator of HebSafeHarbor is not recognized (InvalidParamError).
presidio-analyzer==2.2.31
presidio-anonymizer==2.2.31

hebsafeharbor
