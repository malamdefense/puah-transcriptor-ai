# PII removal dependencies (HebSafeHarbor) — installed in a virtual environment *isolated* (.venv-pii)
# only, because they require pydantic<2 / spaCy 3.2 that conflict with the main application.
# The worker calls them as a sub-process via app/pii_runner.py.
#
# Manual installation:
#   python -m venv .venv-pii
#   .venv-pii\Scripts\python -m pip install -r requirements-pii.txt
# (or simply: python rebuild.py — installs it automatically)

# spaCy 3.2 / thinc / blis old versions built against numpy 1.x — numpy 2 breaks them
# (numpy.dtype size changed). Must be pinned below 2.
numpy<2

# presidio-analyzer pinned to 2.2.31 (the latest version supporting spaCy 3.2). Must be pinned
# also to the same version for the anonymizer — otherwise the custom operator 'replace_in_hebrew'
# of HebSafeHarbor is not recognized (InvalidParamError).
presidio-analyzer==2.2.31
presidio-anonymizer==2.2.31

hebsafeharbor
